          def
          
          
          def name(params) := term; infers the type; def name(params) : T := term; states it; def name(params) : T { … } proves T in a block. Use def for constructions and proofs alike.
            A first proof and files →
          
          
          def four : Nat := 4;

def add_two(n : Nat) := succ(succ(n));

def add_two_to_two : add_two(2) = four {
  exact refl(4);
}
          
        
          
        
          
          
          Parameters and application

            A first proof and files →
          
          
          def pick_second(A, B : U0, x, y : A, b : B) := y;

def picked := pick_second(Nat, Unit, 1, 2, tt);

def from_one := pick_second(Nat, Unit, 1);

          Functions: fun
          
          
          fun (x : A) => body constructs a function, and fun (x, y : A) => body takes several arguments. fun x => body reads the domain from an expected function type.

          def twice(A : U0, f : A -> A) := fun (x : A) => f(f(x));

def add_two := twice(Nat, fun n => succ(n));

          forall x : A. B is a dependent function type (Π): its result type may mention the input. A -> B is the case where it does not.
            Functions, pairs and sums →
          
          
          def apply_at_zero(P : Nat -> U0, f : forall n : Nat. P(n)) : P(0) {
  exact f(0);
}
          
        
          
        
          
          
          Holes and named arguments: _, x := e
          
          
          A hole _ is an argument the other arguments' types or the expected type determine. x := e gives the parameter x; the parameters before it that nothing gives are holes. Click a hole to see its value.
            Functions, pairs and sums →
          
          
          def concat(U < UU0, A : U, x, y, z : A, p : x = y, q : y = z) : x = z := trans(p, q);

def joined(a, b, c : Nat, p : a = b, q : b = c) : a = c := concat(U0, _, _, _, _, p, q);
def named(a, b, c : Nat, p : a = b, q : b = c) : a = c := concat(U0, p := p, q := q);
          
        
          
        
          
          
          Implicit parameters: def f{{A : U}}(…)
          
          
          Parameters in double braces before the parameter list are implicit: a call reads them off its other arguments, universes included. To give them, write them in double braces after the name, in order or by name: concat{{U0, Nat}}(p, q), concat{{A := Nat}}(p, q).
            Functions, pairs and sums →
          
          
          def concat{{U < UU0, A : U, x, y, z : A}}(p : x = y, q : y = z) : x = z := trans(p, q);

def joined(a, b, c : Nat, p : a = b, q : b = c) : a = c := concat(p, q);
def given(a, b, c : Nat, p : a = b, q : b = c) : a = c := concat{{U0, Nat}}(p, q);

          def triple : Nat and Unit and Nat {

def two_exists : exists n : Nat. n = 2 {

def on_the_right : Unit or Nat {

  left u => 0;
  right n => n;

          Proof blocks
          
        
          
          
          intro
          
          
          Introduces the next input of a forall goal, or the assumption of an implication; intro x, y, h; introduces several. Click a line number in the workspace to see the goal.
            Proof blocks →
          
          
          def curry(A, B, C : U0) : ((A and B) -> C) -> A -> B -> C {
  intro f, a, b;
  exact f((a, b));
}
          
        
          
        
          
          
          exact
          
          
          Supplies a term of the goal's type and finishes the block. Like match, rfl, calc and simpa, nothing follows it in its block.
            Proof blocks →
          
          
          def identity_function(A : U0) : A -> A {
  exact fun (a : A) => a;
}
          
        
          
        
          
          
          let
          
          
          let x := term; names a term, let x : T := term; checks it against T, and let x : T { … } proves T in a block. The checker sees through every form.
            Proof blocks →
          
          
          def three_ways(n : Nat) : n = n {
  let p := refl(n);
  let q : n = n := p;
  let r : n = n {
    exact q;
  }
  exact r;
}
          
        
          
        
          
          
          obtain
          
          
          Unpacks a pair or an existential into its parts, with the same tuple shape as (a, b, c).
            Proof blocks →
          
          
          def witness(t : exists n : Nat. (n = n) and Unit) : Nat {
  obtain (n, same, u) := t;
  exact n;
}
          
        
          
        
          
          
          match on a sum
          
          
          Splits a sum into one block per side, each proving the goal at its side.
            Proof blocks →
          
          
          def swap_sum(A, B : U0, value : A or B) : B or A {
  match value {
    left(a) => { exact right(a); }
    right(b) => { exact left(b); }
  }
}
          
        
          
        
          Computation and conversion
          
        
          
          
          Equal by computation
          
          
          Two terms are definitionally equal when they compute to the same normal form: functions apply, definitions unfold, and eliminators compute on constructors. refl and rfl prove exactly these equalities; n + 0 = n for a variable n is not one of them. Older proofs’ with unfolding [name] { expression } is still accepted, and changes nothing.
            Computation and conversion →
          
          
          import naturals;

def double(n : Nat) := n + n;

def double_two : double(2) = 4 {
  rfl;
}

def applied : (fun (n : Nat) => succ(n))(0) = 1 {
  rfl;
}

def small_in_large : U1 := U0;

            Universes →
          
          
          def identity(U < UU0, A : U, x : A) : A := x;

def three := identity(U0, Nat, 3);

def small_identity := identity(U0);
          
        
          
        
          
          
          next and max
          
          
          next(U) is the universe above U, and max(U, V) the larger of two.
            Universes →
          
          
          def type_of(U < UU0) : next(U) := U;

def both(U, V < UU0, A : U, B : V) : max(U, V) := A and B;
          
        
          
        
          Equality and paths
          
        
          
          
          Equality and refl
          
          
          x =[T] y is equality in T, and x = y infers the carrier. refl(x) proves x = x, and any equality whose sides compute to the same value.
            Equality and paths →
          
          
          def explicit_carrier(A : U0, x : A) : x =[A] x {
  exact refl(x);
}

def by_computation : succ(succ(0)) = 2 {
  exact refl(2);
}
          
        
          
        
          
          
          sym, trans and cong
          
          
          sym(p) reverses a path, trans(p, q) joins two, and cong(f, p) maps one through f. They are checked constructions, not axioms.
            Equality and paths →
          
          
          def join_and_map(A, B : U0, f : A -> B, x, y, z : A, p : x = y, q : y = z) : f(z) = f(x) {
  exact sym(cong(f, trans(p, q)));
}
          
        
          
        
          
          
          transport and along
          
          
          transport(C, x, y, p, v) moves v : C(x) along p : x = y into C(y). along C by p from v is the same, with the endpoints read from p.
            Equality and paths →
          
          
          def move(A : U0, C : A -> U0, x, y : A, p : x = y, v : C(x)) : C(y) {
  exact along C by p from v;
}
          
        
          
        
          
          
          Path induction
          
          
          path_induction proves a statement about every path from the case of refl.
            Equality and paths →
          
          
          def symmetric(A : U0, x, y : A, p : x = y) : y = x {
  exact path_induction(A, fun (a, b : A, q : a = b) => b = a,
    fun (a : A) => refl(a), x, y, p);
}

def functions_into_nat(A : U0) : IsSet(U0, A -> Nat) {

          Natural numbers and inductive types
          
        
          
          
          inductive
          
          
          inductive Name(params) : U0 { c; d(x : A, rest : Name(params)); } declares a type by its constructors. A constructor takes only its declared arguments; the type's parameters come from the expected type.
            Natural numbers and inductive types →
          
          
          inductive List(U < UU0, A : U) : U {
  nil;
  cons(head : A, tail : List(U, A));
}

def numbers : List(U0, Nat) := cons(2, cons(3, nil));
          
        
          
        
          
          
          match and structural recursion
          
          
          match n { zero => …; succ(k) => …; } gives one branch per constructor. A definition whose body is the match can call itself on a child such as k. In a proof block, match n { zero => { … } succ(k) => { … } } proves the goal for each constructor. A match can take apart several values, match m, n { … }, with nested patterns, variables and _; the first clause that fits a case takes it.
            Natural numbers and inductive types →
          
          
          def double(n : Nat) : Nat := match n {
  zero => 0;
  succ(k) => succ(succ(double(k)));
};

def double_three : double(3) = 6 { rfl; }
          
        
          
        
          
          
          induction
          
          
          induction v [as z] return C { c(xs, hs) => …; } on any declared type: each clause names the constructor’s arguments, then an induction hypothesis for each recursive argument. In a proof block, induction v { c(xs, hs) => { … } } takes its motive from the goal. Natural numbers also have zero => …; succ h => …;.
            Natural numbers and inductive types →
          
          
          inductive List(U < UU0, A : U) : U {
  nil;
  cons(head : A, tail : List(U, A));
}

def length(xs : List(U0, Nat)) : Nat := induction xs return Nat {
  nil => 0;
  cons(head, tail, ih) => succ(ih);
};
          
        
          
        
          
          
          Binary literals
          
          
          After import binary_naturals;, 0b110 is six in BinaryNat, one constructor per bit. Hover it to see the expansion. Decimal literals stay unary.
            Natural numbers and inductive types →
          
          
          import binary_naturals;

def six : BinaryNat := 0b110;
          
        
          
        
          Equational reasoning and simplification
          
        
          
          
          rfl
          
          
          Closes an equality whose two sides compute to the same value.
            Equational reasoning →
          
          
          import naturals;

def successor_first(n, m : Nat) : succ(n) + m = succ(n + m) {
  rfl;
}
          
        
          
        
          
          
          calc
          
          
          Chains equalities; _ is the previous right side. A step is proved by a term or a nested block.
            Equational reasoning →
          
          
          import naturals;

def through(a, b : Nat, p : a = b) : a + 0 = b {
  calc {
    a + 0 = a by nat_add_zero(a);
    _ = b by p;
  }
}
          
        
          
        
          
          
          rw
          
          
          Rewrites the goal with instantiated equalities, in order; <- rewrites right to left, and at lhs occurrence 2 picks one place. A goal that then holds by computation is closed.
            Equational reasoning →
          
          
          import naturals;

def rewrite_then_finish(n, m : Nat, h : n = m) : n + 0 = m {
  rw [nat_add_zero(n)];   // goal: n = m
  exact h;
}
          
        
          
        
          
          
          simp only
          
          
          Rewrites with the listed rules everywhere, repeatedly, finding their arguments by matching. Click a simp in the workspace to see its trace.
            Equational reasoning →
          
          
          import naturals;

def add_zero_twice(n : Nat) : (n + 0) + 0 = n {
  simp only [nat_add_zero];
}
          
        
          
        
          
          
          simpa
          
          
          Simplifies both the goal and the type of a hypothesis, then uses the hypothesis.
            Equational reasoning →
          
          
          import naturals;

def simpa_example(n, m : Nat, h : n + 0 = m) : n = m + 0 {
  simpa only [nat_add_zero] using h;
}
          
        
          
        
          
          
          simp_rule and simp_set
          
          
          simp_rule registers a default rule and simp_set names a set; both carry through imports. Then simp; uses the defaults, and simp [set] at h as h2; adds a simplified copy of h.
            Equational reasoning →
          
          
          import naturals;

simp_rule nat_add_zero priority 10;
simp_set units := [nat_add_zero];

def by_default(n : Nat) : (n + 0) + 0 = n {
  simp;
}

def simplified_copy(n, m : Nat, h : n + 0 = m) : n = m {
  simp [units] at h as h2;   // h2 : n = m
  exact h2;
}
          
        
          
        
          
          
          Conditional rules
          
          
          A rule with a premise fires when simp with [h]; supplies it.
            Equational reasoning →
          
          
          import naturals;

def zero_if(n : Nat, h : n = 0) : n + 0 = 0 {
  calc {
    n + 0 = n by nat_add_zero(n);
    _ = 0 by h;
  }
}

simp_rule zero_if priority 20;

def conditional(n : Nat, h : n = 0) : n + 0 = 0 {
  simp with [h];
}
          
        
          
        
          Cubical paths and higher inductive types
          
        
          
          
          path i => and p @ i
          
          
          A path is a function of an interval coordinate. Where a path type is expected, path i => body builds one, and p @ i evaluates p at i.
            Cubical paths →
          
          
          def reverse(A : U0, x, y : A, p : x = y) : y = x {
  exact path i => p @ flip(i);
}

def at_the_start(A : U0, x, y : A, p : x = y) : p @ 0 = x {
  exact refl(x);
}
          
        
          
        
          
          
          ext
          
          
          Reduces f = g to f(x) = g(x), without an axiom.
            Cubical paths →
          
          
          def pointwise_path(A, B : U0, f, g : A -> B, h : (forall x : A. f(x) = g(x))) : f = g {
  ext x;
  exact h(x);
}
          
        
          
        
          
          
          Dependent paths: PathP and apd_path
          
          
          PathP(family, a, b) is a path over a family of types. apd_path(f, p) is the dependent path of a dependent function over p, and over C along p by { … } proves one from an equality after transport.
            Cubical paths →
          
          
          def section_path(A : U0, C : A -> U0, f : (forall a : A. C(a)), x, y : A, p : x = y) :
  PathP(fun (i : Interval) => C(p @ i), f(x), f(y)) {
  exact apd_path(f, p);
}
          
        
          
        
          
          
          Composition: comp
          
          
          comp(family, base, walls…) fills from coordinate 0 to 1, starting at the base. A wall face(i, 0, …) fixes the side where another coordinate is 0. With no walls, comp is transport.
            Cubical paths →
          
          
          def join(A : U0, x, y, z : A, p : x = y, q : y = z) : x = z {
  exact path i => comp(fun (j : Interval) => A, p @ i,
    face(i, 0, fun (j : Interval) => x),
    face(i, 1, fun (j : Interval) => q @ j));
}
          
        
          
        
          
          
          Path constructors
          
          
          An inductive declaration can have constructors that build paths, written with their endpoint equality as the result type. A match clause loop @ i => … gives the result along the path, and must agree with the point clauses at its ends.
            Cubical paths →
          
          
          inductive Circle : U0 {
  base;
  loop : base = base;
}

def reverse(c : Circle) : Circle := match c {
  base => base;
  loop @ i => loop @ -i;
};
          
        
          
        
          
          
          Pushouts
          
          
          import pushout; declares Pushout(U, V, W, C, A, B, f, g), which glues A and B along the two maps from C: the path constructor push(c) joins inl(f(c)) to inr(g(c)).
            Cubical paths →
          
          
          import pushout;

def TwoPoints : U0 := Pushout(U0, U0, U0, Nat, Unit, Unit, fun (n : Nat) => tt, fun (n : Nat) => tt);

def west : TwoPoints := inl(tt);

def east : TwoPoints := inr(tt);

def bridge(n : Nat) : west = east {
  exact push(n);
}

def merely(A : U0, a : A) : Truncate(U0, A) {

          computable def
          
          
          Rejects the declaration if it uses an assumption, directly or through a definition.
            Excluded middle, choice and computability →
          
          
          import naturals;

computable def double(n : Nat) := n + n;
          
        
          
        
          
          
          evaluate
          
          
          evaluate term expecting value; checks what a closed term computes to. The term may use no assumption.
            Excluded middle, choice and computability →
          
          
          import naturals;

def double(n : Nat) := n + n;

evaluate double(3) expecting 6;
